1120.Maximum Average Subtree

1# Definition for a binary tree node.
2# class TreeNode:
3# def __init__(self, val=0, left=None, right=None):
4# self.val = val
5# self.left = left
6# self.right = right
7
8class Solution:
9 def dfs(self, root):
10 if not root:
11 return (0, 0)
12
13 leftCount, leftSum = self.dfs(root.left)
14 rightCount, rightSum = self.dfs(root.right)
15
16 currentAvg = (leftSum + rightSum + root.val) / (leftCount + rightCount + 1)
17
18 self.maxAvg = max(currentAvg, self.maxAvg)
19
20 return (leftCount + rightCount + 1, leftSum + rightSum + root.val)
21
22 def maximumAverageSubtree(self, root: Optional[TreeNode]) -> float:
23 self.maxAvg = 0
24 self.dfs(root)
25 return self.maxAvg
